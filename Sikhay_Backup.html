Google Drive Auto-Upload
Upload to Google Drive
Automatic upload sa iyong Google Drive gamit ang Apps Script. One-time setup lang, tapos auto-upload na tuwing mag-e-export.
One-time setup (5 minutes)
  1. Pumunta sa script.google.com → New Project
  2. I-replace ang laman ng editor ng code sa ibaba:
function doPost(e) {
  try {
    var body   = JSON.parse(e.postData.contents);
    var name   = body.filename || ('sikhay-backup-' +
      Utilities.formatDate(new Date(),'Asia/Manila','yyyy-MM-dd-HHmm') + '.json');
    var folder, folders = DriveApp.getFoldersByName('Sikhay Backups');
    folder = folders.hasNext() ? folders.next() : DriveApp.createFolder('Sikhay Backups');
    folder.createFile(name, JSON.stringify(body.data, null, 2), MimeType.PLAIN_TEXT);
    return ContentService.createTextOutput('{"ok":true}')
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput('{"ok":false}')
      .setMimeType(ContentService.MimeType.JSON);
  }
}
